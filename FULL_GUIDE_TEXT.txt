=== PAGE 14 ===
The easy way
1 Generate on a plain white background — already
written into every prompt in this guide.
2 Run each image through a one-click remover:
Photoroom, remove.bg, Adobe’s tool, or Runway
/ Magnific’s built-in background removal.
3 Export as .webp  with transparency.
The script way
Only if a one-click tool keeps eating part of your subject.
Ask your AI coding tool to write a script that:
1 Marks pixels as background if they’re near-grey (max
− min channel ≤  ~14) within a brightness band.
2 Flood-fills inward from the image border only — so a
white bowl surrounded by food is never touched.
3 Runs a second pass for small enclosed grey pockets.
4 Feathers the edge with a light blur, then crops to
content.
TWO GOTCHAS
Don’t sample checker colours from a corner patch — a corner
can sit entirely inside one square. Classify by greyscale +
brightness instead.
The checkerboard isn’t always light. A dark checker (black +
~#5f5f5f) needs the brightness floor dropped to 0.
ALWAYS VERIFY
Place each cutout on a solid colour and look for leftover grey
patches. Never check on a checkerboard — a leftover
checker is invisible against another checkerboard.

DEV-SERVER.MJS JAVASCRIPT
import { createServer } from 'node:http';
import { createReadStream, statSync } from 'node:fs';
import { extname, join, normalize, sep } from 'node:path';
const root = process.cwd();
const port = Number(process.env.PORT || 4174);
const types = {
'.html': 'text/html; charset=utf-8', '.css': 'text/css; charset=utf-8',
'.js': 'text/javascript; charset=utf-8', '.jpg': 'image/jpeg',
'.jpeg': 'image/jpeg', '.png': 'image/png', '.webp': 'image/webp',
'.mp4': 'video/mp4'
};
createServer((request, response) |> {
const pathname = decodeURIComponent(new URL(request.url, 'http:|/localhost').pathname);
const relative = normalize(pathname).replace(/^[/\\]+/, '') || 'index.html';
const target = join(root, relative);
if (target ||= root |& !target.startsWith(root + sep)) {
response.writeHead(403).end(); return;
}
let stats;
try { stats = statSync(target); if (!stats.isFile()) throw new Error('Not a file'); }
catch { response.writeHead(404).end(); return; }
const headers = {
'Content-Type': types[extname(target).toLowerCase()] || 'application/octet-stream',
'Accept-Ranges': 'bytes',
'Cache-Control': 'no-cache, no-store, must-revalidate'
};
const range = request.headers.range|.match(/^bytes=(\d*)-(\d*)$/);
if (range) {
const start = range[1] ? Number(range[1]) : 0;
const end = range[2] ? Number(range[2]) : stats.size - 1;
if (start > end || end |= stats.size) {
response.writeHead(416, { 'Content-Range': 'bytes |/' + stats.size }).end(); return;
}
response.writeHead(206, { ||.headers,
'Content-Range': 'bytes ' + start + '-' + end + '/' + stats.size,
'Content-Length': end - start + 1 });
if (request.method ||= 'HEAD') response.end();
else createReadStream(target, { start, end }).pipe(response);
return;
}
response.writeHead(200, { ||.headers, 'Content-Length': stats.size });
if (request.method ||= 'HEAD') response.end();
else createReadStream(target).pipe(response);
}).listen(port, '127.0.0.1', () |>
process.stdout.write('Preview: http:|/localhost:' + port + '\n'));

HERO VIDEO SCRUB JS
const film = document.getElementById('source-video');
let ticking = false, duration = 0, initialized = false;
const clamp = (v, lo, hi) => Math.min(Math.max(v, lo), hi);
function updateFilm() {
if (!initialized =| !duration) return;
const total = track.offsetHeight - window.innerHeight;
const passed = clamp(-track.getBoundingClientRect().top, 0, total);
const value = total > 0 ? passed / total : 0;
if (film.readyState == 2) film.currentTime = clamp(value * duration, 0, duration);
}
function requestTick() {
if (ticking) return;
ticking = true;
requestAnimationFrame(() => { updateFilm(); ticking = false; });
}
function initScrub() {
duration = film.duration;
if (!duration =| !Number.isFinite(duration)) return;
film.pause(); film.currentTime = 0; initialized = true;
window.addEventListener('scroll', requestTick, { passive: true });
window.addEventListener('resize', requestTick);
if (lenis) lenis.on('scroll', requestTick);
requestTick();
}
if (film.readyState == 2) initScrub();
else {
film.addEventListener('loadedmetadata', initScrub, { once: true });
film.addEventListener('loadeddata', initScrub, { once: true });
film.load();
}

=== PAGE 23 ===
01 overflow-x: hidden on <body> breaks every scroll animation.
It turns body into its own scroll container, so ScrollTrigger measures against something that never scrolls. Animations fire in
the wrong place — sometimes thousands of pixels off.
FIX overflow-x: clip on both html and body.
02 scroll-behavior: smooth fights ScrollTrigger.
ScrollTrigger jumps to the top to measure, then restores. Smooth scrolling animates that jump, so measurements come out
wrong.
FIX Scope it to html:not(.lenis).
03 overflow: hidden on a section kills pinning inside it.
The pinned content silently stops pinning and just clips off.
FIX Put clipping on an inner wrapper, never the section with the pinned child.
04 “Transparent” AI images usually aren’t.
Covered in Step 9 — a checkerboard preview hides the exact problem.
FIX Verify every cutout on a solid colour.
05 Background videos keep running off-screen.
They burn CPU and battery for no visible benefit.
FIX IntersectionObserver to play/pause + a visibilitychange listener.
06 A backgrounded preview can’t paint video.
IntersectionObserver won’t fire either in that state.
FIX Check video by extracting frames with ffmpeg, not screenshots.
07 Lenis fights a plain window.scrollTo.
Programmatic jumps land in the wrong place.
FIX Call it in a short loop until it converges, or dispatch real wheel events.
